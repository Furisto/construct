package main

import (
    "fmt"
    "os"
    "github.com/furisto/construct/shared/keyring"
)

func main() {
    provider := keyring.NewKeyringProvider()
    var err error
    switch os.Args[1] {
    case "set": err = provider.Set(os.Args[2], os.Args[3])
    case "get":
        var value string
        value, err = provider.Get(os.Args[2])
        if err == nil { fmt.Print(value) }
    case "delete": err = provider.Delete(os.Args[2])
    }
    if err != nil { fmt.Fprintln(os.Stderr, err); os.Exit(1) }
}
