package main

import (
	"context"
	"encoding/json"
	"fmt"
	"os"
	"path/filepath"

	"entgo.io/ent/dialect"
	"github.com/furisto/construct/backend/memory"
	"github.com/furisto/construct/backend/memory/modelprovider"
	"github.com/furisto/construct/backend/secret"
)

func main() {
	var input struct {
		Data string            `json:"data"`
		Keys map[string]string `json:"keys"`
	}
	must(json.NewDecoder(os.Stdin).Decode(&input))
	encoded, err := os.ReadFile(filepath.Join(input.Data, "secrets", "encryption_key"))
	must(err)
	handle, err := secret.KeysetFromJSON(string(encoded))
	must(err)
	encryption, err := secret.NewClient(handle)
	must(err)
	db, err := memory.Open(dialect.SQLite, "file:"+filepath.Join(input.Data, "construct.db")+"?mode=ro")
	must(err)
	defer db.Close()
	for name, expected := range input.Keys {
		provider, err := db.ModelProvider.Query().Where(modelprovider.NameEQ(name)).Only(context.Background())
		must(err)
		plaintext, err := encryption.Decrypt(provider.Secret, secret.ModelProviderAssociated(provider.ID))
		must(err)
		var auth struct {
			APIKey string `json:"apiKey"`
		}
		must(json.Unmarshal(plaintext, &auth))
		if auth.APIKey != expected {
			must(fmt.Errorf("stored provider key does not match supplied key"))
		}
	}
	fmt.Printf("[cli/provider/storage] %d providers decrypted from daemon SQLite using file-provider encryption key; exact API-key bytes match (redacted)\n", len(input.Keys))
}

func must(err error) {
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
}
