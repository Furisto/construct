const special = "<>&\u2028\u2029\"\\\n\t é 🚀"

func fill(v reflect.Value, depth int, mode string) {
	if depth > 5 {
		return
	}
	switch v.Type() {
	case reflect.TypeOf(time.Time{}):
		v.Set(reflect.ValueOf(time.Date(2025, 2, 3, 4, 5, 6, 123400000, time.UTC)))
		return
	case reflect.TypeOf(uuid.UUID{}):
		v.Set(reflect.ValueOf(uuid.MustParse("12345678-1234-5678-9abc-123456789abc")))
		return
	case reflect.TypeOf(memory.Decimal{}):
		v.Set(reflect.ValueOf(memory.Decimal{Decimal: decimal.RequireFromString("-123456789012345678901234567890.1234000")}))
		return
	case reflect.TypeOf(json.RawMessage{}):
		v.SetBytes([]byte(`{ "z" : "<>&", "a" : [1.0, true] }`))
		return
	}
	switch v.Kind() {
	case reflect.String:
		if mode == "special" {
			v.SetString(special)
		} else {
			v.SetString("value")
		}
	case reflect.Bool:
		v.SetBool(true)
	case reflect.Int, reflect.Int64:
		v.SetInt(7)
	case reflect.Float64:
		v.SetFloat(1.25)
	case reflect.Pointer:
		v.Set(reflect.New(v.Type().Elem()))
		fill(v.Elem(), depth+1, mode)
	case reflect.Slice:
		v.Set(reflect.MakeSlice(v.Type(), 1, 1))
		fill(v.Index(0), depth+1, mode)
	case reflect.Map:
		v.Set(reflect.MakeMap(v.Type()))
		keys := []string{"z", "a"}
		if mode == "special" {
			keys = []string{"z&", "a<"}
		}
		for _, k := range keys {
			item := reflect.New(v.Type().Elem()).Elem()
			fill(item, depth+1, mode)
			v.SetMapIndex(reflect.ValueOf(k), item)
		}
	case reflect.Struct:
		for i := 0; i < v.NumField(); i++ {
			if v.Field(i).CanSet() {
				fill(v.Field(i), depth+1, mode)
			}
		}
	case reflect.Interface:
		if v.Type() == reflect.TypeOf((*provider.ContentBlock)(nil)).Elem() {
			v.Set(reflect.ValueOf(&provider.TextBlock{Text: special}))
		}
	}
}
func write(name string, value any) {
	b, err := json.Marshal(value)
	if err != nil {
		panic(err)
	}
	if err = os.WriteFile(filepath.Join(os.Args[1], name+".json"), b, 0644); err != nil {
		panic(err)
	}
	fmt.Println(name + ".json")
}

func emptyCollections(v reflect.Value) {
	if v.Type() == reflect.TypeOf(json.RawMessage{}) {
		return
	}
	switch v.Kind() {
	case reflect.Slice:
		v.Set(reflect.MakeSlice(v.Type(), 0, 0))
	case reflect.Map:
		v.Set(reflect.MakeMap(v.Type()))
	case reflect.Struct:
		for i := 0; i < v.NumField(); i++ {
			if v.Field(i).CanSet() {
				emptyCollections(v.Field(i))
			}
		}
	}
}
